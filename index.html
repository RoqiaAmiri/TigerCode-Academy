Relational DB Architecture Schema

-- PostgreSQL / MySQL Schema Definition for TigerCode Academy 
CREATE TABLE Users (
    user_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    role VARCHAR(20) CHECK (role IN ('Student', 'Parent', 'Admin')),
    parent_id UUID REFERENCES Users(user_id),
    xp_points INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE Courses (
    course_id SERIAL PRIMARY KEY,
    title VARCHAR(150) NOT NULL,
    description TEXT,
    difficulty_level VARCHAR(30),
    icon_name VARCHAR(50)
);

CREATE TABLE Lessons (
    lesson_id SERIAL PRIMARY KEY,
    course_id INT REFERENCES Courses(course_id) ON DELETE CASCADE,
    title VARCHAR(150) NOT NULL,
    video_url VARCHAR(255),
    sequence_order INT NOT NULL,
    xp_reward INT DEFAULT 50
);

CREATE TABLE Quizzes (
    quiz_id SERIAL PRIMARY KEY,
    lesson_id INT REFERENCES Lessons(lesson_id),
    title VARCHAR(150) NOT NULL
);

CREATE TABLE Questions (
    question_id SERIAL PRIMARY KEY,
    quiz_id INT REFERENCES Quizzes(quiz_id),
    prompt_text TEXT NOT NULL,
    options_json JSONB NOT NULL,
    correct_answer_index INT NOT NULL
);

CREATE TABLE Progress (
    progress_id SERIAL PRIMARY KEY,
    user_id UUID REFERENCES Users(user_id),
    lesson_id INT REFERENCES Lessons(lesson_id),
    is_completed BOOLEAN DEFAULT FALSE,
    completed_at TIMESTAMP
);

CREATE TABLE Badges (
    badge_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    icon_emoji VARCHAR(10),
    description TEXT
);

CREATE TABLE User_Badges (
    user_id UUID REFERENCES Users(user_id),
    badge_id INT REFERENCES Badges(badge_id),
    awarded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (user_id, badge_id)
);

CREATE TABLE Certificates (
    certificate_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES Users(user_id),
    course_id INT REFERENCES Courses(course_id),
    issued_date DATE DEFAULT CURRENT_DATE
);
